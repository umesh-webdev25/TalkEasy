export const uploadMultipleToS3 = async (files, folder, options = {}) => {
    if (!files || files.length === 0) return [];
    const uploadPromises = files.map((file, index) =>
        uploadToS3(file, folder, {
            ...options,
            metadata: {
                ...options.metadata,
                sortOrder: String(index),
            },
        })
    );

    const results = await Promise.allSettled(uploadPromises);

    console.log("allSettled results =>", results);

    results.forEach((result, index) => {
        if (result.status === "rejected") {
            console.log(`❌ Upload failed for file ${index}:`, result.reason);
        } else {
            console.log(`✅ Upload success for file ${index}:`, result.value);
        }
    });

    return results
        .filter((r) => r.status === "fulfilled")
        .map((r, index) => ({
            ...r.value,
            sortOrder: index,
        }));
};
